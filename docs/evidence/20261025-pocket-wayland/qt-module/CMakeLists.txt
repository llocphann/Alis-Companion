cmake_minimum_required(VERSION 3.25)
project(PocketWayland VERSION 6.12.0 LANGUAGES C CXX)
set(QTBASE_SOURCE "" CACHE PATH "Unmodified or patched QtBase v6.12.0 source")
if(NOT EXISTS "${QTBASE_SOURCE}/src/plugins/platforms/wayland/CMakeLists.txt")
 message(FATAL_ERROR "QTBASE_SOURCE must identify QtBase v6.12.0")
endif()
set(QT_GENERATE_SBOM OFF CACHE BOOL "Private component build" FORCE)
set(QT_SKIP_AUTO_PLUGIN_INCLUSION ON)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS BuildInternals Core CorePrivate Gui GuiPrivate WaylandScannerTools)
qt_prepare_standalone_project()
find_package(Wayland REQUIRED COMPONENTS Client Cursor)
find_package(WaylandScanner REQUIRED)
set(QT_FEATURE_waylandscanner ON)
set(QT_FEATURE_wayland ON)
set(QT_FEATURE_wayland_client OFF)
add_custom_target(sync_headers)
foreach(name generate_docs qch_docs prepare_docs docs html_docs install_docs)
 add_custom_target(${name})
endforeach()
add_subdirectory("${QTBASE_SOURCE}/src/plugins/platforms/wayland" wayland)

qt_internal_create_depends_files()
